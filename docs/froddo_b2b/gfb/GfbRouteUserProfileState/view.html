@override
Widget view(BuildContext context) {
  return Scaffold(
    body: Column(
      children: [
        GsaWidgetAppBar(
          label: GfbRouteUserProfileI18N.appBarTitle.value.display,
        ),
        Expanded(
          child: Row(
            children: [
              Expanded(
                child: DecoratedBox(
                  decoration: BoxDecoration(
                    border: Border(
                      right: Theme.of(context).dimensions.smallScreen
                          ? BorderSide.none
                          : BorderSide(
                              color: Theme.of(context).dividerColor,
                            ),
                    ),
                  ),
                  child: ListView(
                    padding: Theme.of(context).listViewPadding,
                    children: [
                      Row(
                        children: [
                          CircleAvatar(
                            child: Icon(
                              Icons.person,
                              color: Colors.white,
                            ),
                          ),
                          const SizedBox(width: 10),
                          Flexible(
                            child: Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: [
                                if (GsaDataUser.instance.user?.personalDetails?.formattedName != null) ...[
                                  GsaWidgetText(
                                    GsaDataUser.instance.user!.personalDetails!.formattedName!,
                                    interpolated: true,
                                    style: Theme.of(context).textTheme.titleLarge,
                                  ),
                                ],
                                if (GfbDataResponse.instance.user?.role?.name != null)
                                  GsaWidgetText.rich(
                                    [
                                      GsaWidgetTextSpan(
                                        '${GfbDataResponse.instance.user?.role?.name?.toUpperCase()}',
                                        interpolated: true,
                                      ),
                                      if (GfbDataResponse.instance.user?.id != null)
                                        GsaWidgetTextSpan(
                                          ' #${GfbDataResponse.instance.user!.id!}',
                                          interpolated: true,
                                          style: TextStyle(
                                            color: Colors.grey,
                                          ),
                                        ),
                                    ],
                                    style: TextStyle(
                                      fontWeight: FontWeight.w300,
                                      fontSize: 10,
                                    ),
                                  ),
                              ],
                            ),
                          ),
                        ],
                      ),
                      if (1 == 2) ...[
                        const Divider(height: 20),
                        for (final userData in <({
                          String label,
                          String value,
                        })>{
                          if (GfbDataResponse.instance.user?.distributor?.name != null)
                            (
                              label: GfbRouteUserProfileI18N.distributorLabel.value.display,
                              value: GfbDataResponse.instance.user!.distributor!.name!,
                            ),
                          if (GfbDataResponse.instance.user?.language?.name != null)
                            (
                              label: GfbRouteUserProfileI18N.languageLabel.value.display,
                              value: GfbDataResponse.instance.user!.language!.name!,
                            ),
                          if (GfbDataResponse.instance.user?.region?.name != null)
                            (
                              label: GfbRouteUserProfileI18N.regionLabel.value.display,
                              value: GfbDataResponse.instance.user!.region!.name!,
                            ),
                        }.indexed) ...[
                          if (userData.$1 != 0) const SizedBox(height: 4),
                          GsaWidgetText.rich(
                            [
                              GsaWidgetTextSpan(
                                '${userData.$2.label}: ',
                                style: TextStyle(
                                  color: Colors.grey,
                                ),
                              ),
                              GsaWidgetTextSpan(
                                userData.$2.value,
                                interpolated: true,
                              ),
                            ],
                          ),
                        ],
                      ],
                      const SizedBox(height: 20),
                      for (final action in <({
                        String label,
                        int value,
                        IconData icon,
                        GsaRoute Function() route,
                      })>{
                        (
                          label: GfbRouteUserProfileI18N.viewClientsButtonLabel.value.display,
                          value: GfbDataRequest.instance.clientRegisters.length + GfbDataResponse.instance.clients.length,
                          icon: Icons.group,
                          route: () {
                            return const GfbRouteClients();
                          },
                        ),
                        (
                          label: GfbRouteUserProfileI18N.viewOrdersButtonLabel.value.display,
                          value: GfbDataResponse.instance.registeredOrders.length + GfbDataRequest.instance.checkoutRegisters.length,
                          icon: Icons.shopping_basket,
                          route: () {
                            return const GfbRouteOrders();
                          },
                        ),
                      }.indexed) ...[
                        if (action.$1 != 0) const SizedBox(height: 14),
                        SizedBox(
                          width: MediaQuery.of(context).size.width,
                          child: GsaWidgetButton.outlined(
                            labelWidget: GsaWidgetText.rich(
                              [
                                GsaWidgetTextSpan(
                                  action.$2.label,
                                ),
                                GsaWidgetTextSpan(
                                  ' (${action.$2.value})',
                                  interpolated: true,
                                ),
                              ],
                            ),
                            icon: action.$2.icon,
                            foregroundColor: Theme.of(context).dimensions.smallScreen || action.$1 == _selectedViewTab
                                ? Theme.of(context).primaryColor
                                : Colors.grey,
                            onTap: Theme.of(context).dimensions.smallScreen
                                ? () async {
                                    await action.$2.route().push();
                                    setState(() {});
                                  }
                                : () {
                                    _selectViewTab(action.$1);
                                  },
                          ),
                        ),
                      ],
                      const SizedBox(height: 14),
                      _WidgetSyncSection(
                        synchronise: () async {
                          _syncTimer?.cancel();
                          _syncTimer = null;
                          await _dataSync();
                          _setSyncTimer();
                        },
                      ),
                      const SizedBox(height: 14),
                      GsaWidgetButton.outlined(
                        label: GfbRouteUserProfileI18N.scannerSetupButtonLabel.value.display,
                        icon: Icons.barcode_reader,
                        onTap: () async {
                          GsaRoute.state<GfbRouteScanState>()?.setDisplayEnabled(true);
                        },
                      ),
                      const SizedBox(height: 14),
                      GsaWidgetButton.outlined(
                        label: GfbRouteUserProfileI18N.logoutButtonLabel.value.display,
                        icon: Icons.logout,
                        onTap: () async {
                          final confirmed = await GsaWidgetOverlayConfirmation(
                            GfbRouteUserProfileI18N.logoutButtonOverlayConfirmationDescription.value.display,
                          ).openDialog();
                          if (confirmed) {
                            await GsaDataUser.instance.logout();
                          }
                        },
                      ),
                    ],
                  ),
                ),
              ),
              if (Theme.of(context).dimensions.largeScreen)
                Expanded(
                  flex: 2,
                  child: _selectedViewTab == 0 ? const GfbRouteClients() : const GfbRouteOrders(),
                ),
            ],
          ),
        ),
      ],
    ),
  );
}
