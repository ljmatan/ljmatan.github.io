factory GfbModelResponseSaleItem.fromJson(Map<String, dynamic> json) {
  return GfbModelResponseSaleItem(
    id: json['id'] ?? json['pro_id'],
    productCode: json['product_code'],
    productName: json['product_name'],
    productDescription: json['product_description'],
    alfaCode: json['alfa_code'],
    forUsers: json['for_users'],
    availableSizes: json['available_sizes'],
    brand: json['brand'],
    mold: json['mold'],
    boys: json['boys'],
    girls: json['girls'],
    unisex: json['unisex'],
    woman: json['woman'],
    man: json['man'],
    tags: json['tags'] == null
        ? []
        : List<String>.from(
            json['tags']!.map((x) => x),
          ),
    images: json['images'] == null
        ? []
        : List<GfbModelResponseSaleItemImage>.from(
            json['images']!.map(
              (x) => GfbModelResponseSaleItemImage.fromJson(x),
            ),
          ),
    icons: json['icons'] == null ? [] : List<String>.from(json['icons']!.map((x) => x)),
    attributes: json['attributes'] == null
        ? []
        : List<GfbModelResponseSaleItemAttribute>.from(
            json['attributes']!.map(
              (x) => GfbModelResponseSaleItemAttribute.fromJson(x),
            ),
          ),
    productPerRegions: json['product_per_regions'] == null
        ? []
        : List<GfbModelResponseSaleItemProductPerRegion>.from(
            json['product_per_regions']!.map(
              (x) => GfbModelResponseSaleItemProductPerRegion.fromJson(x),
            ),
          ),
  );
}
