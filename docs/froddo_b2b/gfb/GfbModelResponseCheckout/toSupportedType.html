GsaModelOrderDraft toSupportedType() {
  final saleItemIds = <String>{
    if (saleItems != null)
      for (final saleItem in saleItems!)
        if (saleItem.productId != null) saleItem.productId!.toString(),
  };
  final cartItems = <GsaModelSaleItem>[];
  for (final saleItemId in saleItemIds) {
    final matchingItem = saleItems?.firstWhereOrNull(
      (saleItem) {
        return saleItem.productId?.toString() == saleItemId;
      },
    );
    if (matchingItem == null) continue;
    cartItems.add(
      GsaModelSaleItem(
        id: saleItemId,
        name: matchingItem.name,
        productCode: matchingItem.code,
        options: [
          if (saleItems != null)
            for (final saleItem in saleItems!)
              if (saleItem.productId?.toString() == saleItemId) saleItem.toSupportedType(),
        ],
      ),
    );
  }
  final itemCount = [
    if (saleItems != null)
      for (final saleItem in saleItems!)
        if (saleItem.optionId != null && saleItem.code != null && saleItem.count != null)
          GsaModelOrderDraftItemCount(
            id: saleItem.productId!.toString(),
            optionId: saleItem.optionId!,
            count: saleItem.count!,
          ),
  ];
  return GsaModelOrderDraft(
    client: client?.toSupportedType()?..id = null,
    items: cartItems,
  )..itemCount = itemCount;
}
