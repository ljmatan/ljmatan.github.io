factory GfbModelResponseCheckout.fromJson(Map json) {
  return GfbModelResponseCheckout(
    id: json['oruse_id'],
    client: json['user'] == null
        ? null
        : GfbModelResponseClient.fromJson(
            json['user'],
          ),
    region: json['region'] == null
        ? null
        : GfbModelResponseRegion.fromJson(
            json['region'],
          ),
    paymentMethod: json['payment_method'] == null
        ? null
        : GfbModelResponseCheckoutPaymentMethod.fromJson(
            json['payment_method'],
          ),
    deliveryMethod: json['delivery_method'] == null
        ? null
        : GfbModelResponseDeliveryMethod.fromJson(
            json['delivery_method'],
          ),
    legalText: json['legal_text'] == null
        ? null
        : GfbModelResponseCheckoutLegalText.fromJson(
            json['legal_text'],
          ),
    orderType: json['order_type'] == null
        ? null
        : GfbModelResponseCheckoutOrderType.fromJson(
            json['order_type'],
          ),
    user: json['agent'] == null
        ? null
        : GfbModelResponseUser.fromJson(
            json['agent'],
          ),
    comment: json['comment'],
    deliveryAddress: json['delivery_address'] == null
        ? null
        : GfbModelResponseClientDeliveryAddress.fromJson(
            json['delivery_address'],
          ),
    total: json['total'],
    deliveryCost: json['delivery_cost'],
    status: json['status'],
    orderConfirmed: json['order_confirmed'],
    dateOfOrder: json['date_of_order'] == null
        ? null
        : DateTime.parse(
            json['date_of_order'],
          ),
    saleItems: json['products'] == null
        ? null
        : List<GfbModelResponseCheckoutSaleItem>.from(
            json['products']!.map(
              (x) {
                return GfbModelResponseCheckoutSaleItem.fromJson(x);
              },
            ),
          ),
  );
}
