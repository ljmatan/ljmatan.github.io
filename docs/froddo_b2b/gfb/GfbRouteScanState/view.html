@override
Widget view(BuildContext context) {
  if (!_displayEnabled) return const SizedBox();
  return Scaffold(
    key: _viewKey,
    body: Column(
      children: [
        GsaWidgetAppBar(
          label: widget.displayName,
          onBackPressed: () {
            setDisplayEnabled(false);
          },
        ),
        Expanded(
          child: Builder(
            builder: (context) {
              if (_disposing) {
                return _WidgetLoadingIndicator(
                  GfbRouteScanI18N.clearingResourcesMessage.value.display,
                );
              }

              if (!_permissionsGranted) {
                return FutureBuilder(
                  future: _getPermissionsStatus(),
                  builder: (context, snapshot) {
                    if (snapshot.connectionState != ConnectionState.done) {
                      return _WidgetLoadingIndicator(
                        GfbRouteScanI18N.permissionRequestedMessage.value.display,
                      );
                    }

                    if (snapshot.hasError || !_permissionsGranted) {
                      return Center(
                        child: GsaWidgetError(
                          snapshot.error?.toString() ?? GfbRouteScanI18N.bluetoothPermissionNotGrantedMessage.value.display,
                          retryLabel: snapshot.error != null ? null : GfbRouteScanI18N.openSettingsButtonLabel.value.display,
                          retry: snapshot.error != null
                              ? () {
                                  setState(() {
                                    _disposeResources();
                                  });
                                }
                              : () async {
                                  await permission_handler.openAppSettings();
                                },
                        ),
                      );
                    }

                    WidgetsBinding.instance.addPostFrameCallback(
                      (_) {
                        setState(() {});
                      },
                    );

                    return const SizedBox();
                  },
                );
              }

              if (!_serviceEnabled) {
                return FutureBuilder(
                  future: _getServiceStatus(),
                  builder: (context, snapshot) {
                    if (snapshot.connectionState != ConnectionState.done) {
                      return _WidgetLoadingIndicator(
                        GfbRouteScanI18N.checkingServiceStatusMessage.value.display,
                      );
                    }

                    if (snapshot.hasError || !_serviceEnabled) {
                      return Center(
                        child: GsaWidgetError(
                          snapshot.error?.toString() ?? GfbRouteScanI18N.bluetoothServiceNotEnabledMessage.value.display,
                          retryLabel: snapshot.error != null ? null : GfbRouteScanI18N.openSettingsButtonLabel.value.display,
                          retry: snapshot.error != null
                              ? () {
                                  setState(() {
                                    _disposeResources();
                                  });
                                }
                              : () async {
                                  await permission_handler.openAppSettings();
                                },
                        ),
                      );
                    }

                    WidgetsBinding.instance.addPostFrameCallback(
                      (_) {
                        setState(() {});
                      },
                    );

                    return const SizedBox();
                  },
                );
              }

              if (!_barcodeSetNameModeSelected) {
                return _WidgetLoadingIndicator(
                  GfbRouteScanI18N.setNameModeTitle.value.display,
                  troubleshootTips: [
                    (
                      message: GfbRouteScanI18N.setNameModeMessage.value.display,
                      barcodeValue: _barcodeValues.setNameMode,
                    ),
                  ],
                  action: (
                    label: GfbRouteScanI18N.continueButtonLabel.value.display,
                    onTap: () {
                      setState(() {
                        _barcodeSetNameModeSelected = true;
                      });
                    },
                  ),
                );
              }

              if (!_barcodeScannerNameSet) {
                return _WidgetLoadingIndicator(
                  GfbRouteScanI18N.setDeviceIdentifierTitle.value.display,
                  troubleshootTips: [
                    (
                      message: GfbRouteScanI18N.setDeviceIdentifierMessage.value.display,
                      barcodeValue: _barcodeScannerName + ' BLE',
                    ),
                  ],
                  action: (
                    label: GfbRouteScanI18N.continueButtonLabel.value.display,
                    onTap: () {
                      setState(() {
                        _barcodeScannerNameSet = true;
                      });
                    },
                  ),
                );
              }

              if (!_barcodeScannerBleModeSet) {
                return _WidgetLoadingIndicator(
                  GfbRouteScanI18N.setScannerModeTitle.value.display,
                  troubleshootTips: [
                    (
                      message: GfbRouteScanI18N.setScannerModeMessage.value.display,
                      barcodeValue: _barcodeValues.setBleMode,
                    ),
                  ],
                  action: (
                    label: GfbRouteScanI18N.continueButtonLabel.value.display,
                    onTap: () {
                      setState(() {
                        _barcodeScannerBleModeSet = true;
                      });
                    },
                  ),
                );
              }

              if (_barcodeScanner == null) {
                return FutureBuilder(
                  future: _scanBleDevices(),
                  builder: (context, snapshot) {
                    if (snapshot.connectionState != ConnectionState.done) {
                      return _WidgetLoadingIndicator(
                        GfbRouteScanI18N.searchingForScannerDeviceMessage.value.display,
                      );
                    }

                    if (snapshot.hasError) {
                      return Center(
                        child: GsaWidgetError(
                          snapshot.error?.toString() ?? GfbRouteScanI18N.errorScanningForBleDevicesMessage.value.display,
                          retry: () {
                            setState(() {
                              _disposeResources();
                            });
                          },
                        ),
                      );
                    }

                    return _WidgetLoadingIndicator(
                      GfbRouteScanI18N.connectingToScannerTitle.value.display,
                      troubleshootTips: [
                        (
                          message: GfbRouteScanI18N.connectingToScannerMessage.value.display,
                          barcodeValue: _barcodeValues.setBleMode,
                        ),
                      ],
                      action: kIsWeb
                          ? (
                              label: GfbRouteScanI18N.browserSetupButtonLabel.value.display,
                              onTap: () {
                                setState(() {
                                  _viewKey = UniqueKey();
                                });
                              },
                            )
                          : null,
                    );
                  },
                );
              }

              if (!_barcodeScannerConnected) {
                return FutureBuilder(
                  future: _connectToBarcodeScanner(),
                  builder: (context, snapshot) {
                    if (snapshot.connectionState != ConnectionState.done) {
                      return _WidgetLoadingIndicator(
                        GfbRouteScanI18N.connectingToScannerTitle.value.display,
                        troubleshootTips: [
                          (
                            message: GfbRouteScanI18N.connectingToScannerTroubleshootSetMode.value.display,
                            barcodeValue: _barcodeValues.setBleMode,
                          ),
                          (
                            message: GfbRouteScanI18N.connectingToScannerTroubleshootReset.value.display,
                            barcodeValue: _barcodeValues.reset,
                          ),
                        ],
                      );
                    }

                    if (snapshot.hasError) {
                      return Center(
                        child: GsaWidgetError(
                          snapshot.error?.toString() ?? GfbRouteScanI18N.genericErrorMessage.value.display,
                          retry: () {
                            setState(() {
                              _disposeResources();
                            });
                          },
                        ),
                      );
                    }

                    return Center(
                      child: Padding(
                        padding: Theme.of(context).listViewPadding,
                        child: GsaWidgetText(
                          GfbRouteScanI18N.establishingDeviceConnectionMessage.value.display,
                        ),
                      ),
                    );
                  },
                );
              }

              if (_scanCharacteristic == null) {
                return FutureBuilder(
                  future: _setupService(),
                  builder: (context, snapshot) {
                    if (snapshot.connectionState != ConnectionState.done) {
                      return _WidgetLoadingIndicator(
                        GfbRouteScanI18N.settingUpMessage.value.display,
                      );
                    }

                    if (snapshot.hasError) {
                      return Center(
                        child: GsaWidgetError(
                          snapshot.error?.toString() ?? GfbRouteScanI18N.genericErrorMessage.value.display,
                        ),
                      );
                    }

                    WidgetsBinding.instance.addPostFrameCallback(
                      (_) {
                        setState(() {});
                      },
                    );

                    return const SizedBox();
                  },
                );
              }

              return const _WidgetSuccess();
            },
          ),
        ),
      ],
    ),
  );
}
