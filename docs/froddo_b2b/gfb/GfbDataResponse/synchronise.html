Future<void> synchronise(
  GfbModelResponseUser userOriginData, {
  bool login = true,
}) async {
  // Fetch and assign delivery method data.
  final deliveryMethodsResponse = await GfbApiCheckout.instance.getDeliveryMethods(
    GfbModelRequestDeliveryMethods(
      regionId: userOriginData.region?.id,
      languageId: userOriginData.language?.id,
    ),
  );
  deliveryMethods = deliveryMethodsResponse;
  // Fetch and assign sale item data.
  final products = await GfbApiProducts.instance.getSaleItems(
    GfbModelRequestProductsSearch(
      distributorId: userOriginData.distributor?.id,
      regionId: userOriginData.region?.id,
      languageId: userOriginData.language?.id,
    ),
  );
  GsaDataSaleItems.instance.collection.clear();
  GsaDataSaleItems.instance.collection.addAll(products);
  // Get active clients and assign the relevant data.
  final clients = await GfbApiClients.instance.getClientList(
    GfbModelRequestClientsSearch(
      distributorId: userOriginData.distributor?.id,
    ),
  );
  GsaDataClients.instance.collection.clear();
  GsaDataClients.instance.collection.addAll(clients);
  // Get user order history.
  final registeredOrdersResponse = await GfbApiCheckout.instance.searchOrders(
    GfbModelRequestCheckoutSearch(
      userId: userOriginData.id,
    ),
  );
  registeredOrders = registeredOrdersResponse;
  // Cache the available response data.
  try {
    await cacheUserData();
    await cacheDeliveryMethodData();
    await cacheSaleItemData();
    await cacheClientsData();
    await cacheRegisteredOrderData();
    responseSyncTime = DateTime.now();
    responseSyncSuccess = true;
    await cacheResponseSyncTime();
    await cacheResponseSyncSuccess();
  } catch (e) {
    if (login) await clearCachedData();
    responseSyncTime = DateTime.now();
    responseSyncSuccess = false;
    await cacheResponseSyncTime();
    await cacheResponseSyncSuccess();
    rethrow;
  }
}
